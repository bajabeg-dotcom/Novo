@echo off
setlocal
cd /d "%~dp0"

where py >nul 2>&1
if not errorlevel 1 (
    set "PYTHON=py -3"
) else (
    where python >nul 2>&1
    if errorlevel 1 (
        echo Python 3.11 or newer was not found.
        exit /b 1
    )
    set "PYTHON=python"
)

if not exist "data\pa800-enhancer.db" (
    %PYTHON% -m pa800_enhancer database init data\pa800-enhancer.db
    if errorlevel 1 exit /b 1
)

%PYTHON% -m pa800_enhancer ui %*
exit /b %errorlevel%
